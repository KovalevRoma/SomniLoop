cmake_minimum_required(VERSION 3.18)
project(somniloop_live2d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CUBISM_SDK "" CACHE PATH "Official Cubism SDK for Native root")
if(NOT EXISTS "${CUBISM_SDK}/Core/include/Live2DCubismCore.h")
  message(FATAL_ERROR "Set CUBISM_SDK to the extracted official Native SDK")
endif()
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
set(FRAMEWORK_SOURCE OpenGL)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(FRAMEWORK_DEFINITIOINS CSM_TARGET_LINUX_GL)
  set(CUBISM_CORE "${CUBISM_SDK}/Core/lib/linux/x86_64/libLive2DCubismCore.a" CACHE FILEPATH "Cubism Core library")
elseif(APPLE)
  set(FRAMEWORK_DEFINITIOINS CSM_TARGET_MAC_GL)
  set(CUBISM_CORE "${CUBISM_SDK}/Core/lib/macos/libLive2DCubismCore.a" CACHE FILEPATH "Cubism Core library")
else()
  message(FATAL_ERROR "This build recipe supports desktop Linux and macOS OpenGL; no Windows APIs are used")
endif()
add_subdirectory("${CUBISM_SDK}/Framework" framework)
target_link_libraries(Framework PUBLIC GLEW::GLEW OpenGL::GL)
add_library(somniloop_live2d SHARED bridge.cpp)
target_include_directories(somniloop_live2d PRIVATE "${CUBISM_SDK}/Core/include")
target_compile_definitions(somniloop_live2d PRIVATE ${FRAMEWORK_DEFINITIOINS})
target_link_libraries(somniloop_live2d PRIVATE Framework "${CUBISM_CORE}" GLEW::GLEW OpenGL::GL)
target_compile_options(somniloop_live2d PRIVATE -Wall -Wextra -Wpedantic)
